   119→/** Normalisasi nomor WA menjadi format internasional untuk tautan wa.me. */
   120→function waDigits(raw: string): string {
   121→  let d = (raw || "").replace(/\D/g, "");
   122→  if (d.startsWith("0")) d = `62${d.slice(1)}`;
   123→  else if (d.startsWith("8")) d = `62${d}`;
   124→  return d;
   125→}

   139→/** Form "Berlangganan Kabar" → POST /api/subscribers {email}. Sukses: toast; gagal: inline. */
   140→function NewsletterForm() {
   141→  const { t } = useT();
   142→  const { toast } = useToast();
   143→  const [email, setEmail] = useState("");
   144→  const [loading, setLoading] = useState(false);
   145→  const [msg, setMsg] = useState<NewsMsg | null>(null);
   146→
   147→  const submit = async (e: React.FormEvent) => {
   148→    e.preventDefault();
   149→    setMsg(null);
   150→    const value = email.trim();
   151→    if (!EMAIL_RE.test(value)) {
   152→      setMsg({ kind: "error", text: t("footer.newsErrInvalid") });
   153→      return;
   154→    }
   155→    setLoading(true);
   156→    try {
   157→      // Kontrak: 201 {ok:true} / duplikat 201 {ok:true,already:true} / 409 — semuanya ramah.
   158→      const res = await apiSend<{ ok?: boolean; already?: boolean }>("/api/subscribers", "POST", { email: value });
   159→      if (res?.already) {
   160→        setMsg({ kind: "dup", text: t("footer.newsDupMsg") });
   161→      } else {
   162→        toast({ title: t("footer.newsOkTitle"), description: t("footer.newsOkDesc") });
   163→        setEmail("");
   164→      }
   165→    } catch (err) {
   166→      const raw = ((err as Error).message || "").trim();
   167→      if (/sudah|terdaftar|already|exist|409/i.test(raw)) {
   168→        setMsg({ kind: "dup", text: t("footer.newsDupMsg") });
   169→      } else {
   170→        setMsg({ kind: "error", text: raw || t("footer.newsErrMsg") });
   171→      }
   172→    } finally {
   173→      setLoading(false);
   174→    }
   175→  };
   176→
   177→  return (
   178→    <div className="rounded-2xl border border-white/10 bg-white/5 p-5 sm:p-6 shadow-[0_18px_50px_-24px_rgba(0,0,0,0.6)] flex flex-col lg:flex-row lg:items-center gap-4">
   179→      <div className="flex items-start gap-3 flex-1">
   180→        <span className="shrink-0 h-10 w-10 rounded-xl grid place-items-center bg-gold/15 text-gold ring-1 ring-gold/30">
   181→          <Icon name="bell" className="h-5 w-5" />
   182→        </span>
   183→        <div>
   184→          <h3 className="font-semibold text-white text-sm tracking-wide">{t("footer.newsTitle")}</h3>
   185→          <p className="mt-1 text-xs leading-relaxed text-emerald-100/70">{t("footer.newsDesc")}</p>
   186→        </div>
   187→      </div>
   188→      <div className="w-full lg:max-w-md">
   189→        <form onSubmit={submit} noValidate className="flex flex-col sm:flex-row gap-2">
   190→          <label htmlFor="newsletter-email" className="sr-only">
   191→            {t("footer.newsAria")}
   192→          </label>
   193→          <Input
   194→            id="newsletter-email"
   195→            type="email"
   196→            value={email}
   197→            onChange={(e) => setEmail(e.target.value)}
   198→            placeholder={t("footer.newsPlaceholder")}
   199→            aria-label={t("footer.newsAria")}
   200→            aria-invalid={msg?.kind === "error"}
   201→            autoComplete="email"
   202→            dir="ltr"
   203→            className="flex-1 bg-white/10 border-white/20 text-white placeholder:text-emerald-100/40"
   204→          />
   205→          <Button
   206→            type="submit"
   207→            disabled={loading}
   208→            className="bg-gold text-forest-deep font-bold hover:bg-gold/90 shrink-0"
   209→          >
   210→            {loading ? (
   211→              <Icon name="loader-2" className="h-4 w-4 me-1.5 animate-spin" />
   212→            ) : (
   213→              <Icon name="send" className="h-4 w-4 me-1.5" />
   214→            )}
   215→            {t("footer.newsBtn")}
   216→          </Button>
   217→        </form>
   218→        {msg && (
   219→          <p
   220→            role="status"
   221→            className={
   222→              msg.kind === "dup"
   223→                ? "mt-2 text-xs font-semibold text-emerald-100/80"
   224→                : "mt-2 text-xs font-semibold text-gold"
   225→            }
   226→          >
   227→            {msg.text}
   228→          </p>
   229→        )}
   230→      </div>
   231→    </div>
   232→  );
   233→}
